module bridge(
 input logic clk,rst_n,
 input logic [31:0] s_awaddr, input logic s_awvalid, output logic s_awready,
 input logic [31:0] s_wdata, input logic [3:0] s_wstrb, input logic s_wvalid, output logic s_wready,
 output logic [1:0] s_bresp, output logic s_bvalid, input logic s_bready,
 input logic [31:0] s_araddr, input logic s_arvalid, output logic s_arready,
 output logic [31:0] s_rdata, output logic [1:0] s_rresp, output logic s_rvalid, input logic s_rready,
 output logic [31:0] paddr,pwdata, output logic [3:0] pstrb, output logic [2:0] pprot,
 output logic [2:0] psel, output logic penable,pwrite, input logic [95:0] prdata, input logic [2:0] pready,pslverr
);

 localparam [2:0] ST_IDLE=3'd0, ST_SETUP=3'd1, ST_ACCESS=3'd2, ST_BRESP=3'd3, ST_RRESP=3'd4, ST_UNMAP_W=3'd5;

 logic [2:0] state;
 logic aw_have, w_have;
 logic [31:0] aw_addr, aw_wdata;
 logic [3:0] aw_strb;
 logic [2:0] target;
 logic [31:0] local_addr;
 logic [1:0] resp_reg;
 logic [31:0] rdata_reg;

 function automatic [2:0] dec(input logic [31:0] a);
   if (a >= 32'h00021000 && a <= 32'h00021fff) dec = 3'd1;
   else if (a >= 32'h00001000 && a <= 32'h00002fff) dec = 3'd2;
   else if (a >= 32'h80010000 && a <= 32'h80012fff) dec = 3'd4;
   else dec = 3'd0;
 endfunction

 function automatic [31:0] off(input logic [31:0] a, input logic [2:0] t);
   if (t == 3'd1) off = a - 32'h00021000;
   else if (t == 3'd2) off = a - 32'h00001000;
   else off = a - 32'h80010000;
 endfunction

 assign s_awready = (state == ST_IDLE) && !aw_have;
 assign s_wready  = (state == ST_IDLE) && !w_have;

 always_ff @(posedge clk) begin
   if (!rst_n) begin
     state <= ST_IDLE;
     aw_have <= 1'b0;
     w_have <= 1'b0;
     aw_addr <= 32'b0;
     aw_wdata <= 32'b0;
     aw_strb <= 4'b0;
     target <= 3'b0;
     local_addr <= 32'b0;
     resp_reg <= 2'b0;
     rdata_reg <= 32'b0;
     s_bvalid <= 1'b0;
     s_rvalid <= 1'b0;
     s_bresp <= 2'b0;
     s_rresp <= 2'b0;
     s_rdata <= 32'b0;
     psel <= 3'b0;
     penable <= 1'b0;
     paddr <= 32'b0;
     pwdata <= 32'b0;
     pstrb <= 4'b0;
     pwrite <= 1'b0;
   end else begin
     if ((state == ST_IDLE) && s_awvalid && s_awready) begin
       aw_have <= 1'b1;
       aw_addr <= s_awaddr;
     end
     if ((state == ST_IDLE) && s_wvalid && s_wready) begin
       w_have <= 1'b1;
       aw_wdata <= s_wdata;
       aw_strb <= s_wstrb;
     end

     case (state)
       ST_IDLE: begin
         psel <= 3'b0;
         penable <= 1'b0;
         if (aw_have && w_have) begin
           target <= dec(aw_addr);
           local_addr <= off(aw_addr, dec(aw_addr));
           if (dec(aw_addr) != 3'd0) begin
             state <= ST_SETUP;
             psel <= dec(aw_addr);
             penable <= 1'b0;
             paddr <= off(aw_addr, dec(aw_addr));
             pwdata <= aw_wdata;
             pstrb <= aw_strb;
             pwrite <= 1'b1;
           end else begin
             state <= ST_UNMAP_W;
             s_bvalid <= 1'b1;
             s_bresp <= 2'b11;
             aw_have <= 1'b0;
             w_have <= 1'b0;
           end
         end else if (s_arvalid) begin
           target <= dec(s_araddr);
           local_addr <= off(s_araddr, dec(s_araddr));
           if (dec(s_araddr) != 3'd0) begin
             state <= ST_SETUP;
             psel <= dec(s_araddr);
             penable <= 1'b0;
             paddr <= off(s_araddr, dec(s_araddr));
             pwrite <= 1'b0;
             pstrb <= 4'b0;
           end else begin
             state <= ST_RRESP;
             s_rvalid <= 1'b1;
             s_rresp <= 2'b11;
             s_rdata <= 32'b0;
           end
         end
       end
       ST_SETUP: begin
         penable <= 1'b1;
         state <= ST_ACCESS;
       end
       ST_ACCESS: begin
         if (pready[target[2:1]] || (target==3'd1 && pready[0]) || (target==3'd2 && pready[1]) || (target==3'd4 && pready[2])) begin
           // dummy
         end
         if (((target==3'd1) && pready[0]) || ((target==3'd2) && pready[1]) || ((target==3'd4) && pready[2])) begin
           psel <= 3'b0;
           penable <= 1'b0;
           if (pwrite) begin
             s_bvalid <= 1'b1;
             s_bresp <= pslverr[0] ? 2'b10 : 2'b00;
             state <= ST_BRESP;
           end else begin
             s_rvalid <= 1'b1;
             s_rresp <= pslverr[0] ? 2'b10 : 2'b00;
             if (target == 3'd1) s_rdata <= prdata[31:0];
             else if (target == 3'd2) s_rdata <= prdata[63:32];
             else s_rdata <= prdata[95:64];
             state <= ST_RRESP;
           end
           aw_have <= 1'b0;
           w_have <= 1'b0;
         end
       end
       ST_BRESP: begin
         if (s_bready) begin
           s_bvalid <= 1'b0;
           state <= ST_IDLE;
         end
       end
       ST_RRESP: begin
         if (s_rready) begin
           s_rvalid <= 1'b0;
           state <= ST_IDLE;
         end
       end
       ST_UNMAP_W: begin
         if (s_bready) begin
           s_bvalid <= 1'b0;
           state <= ST_IDLE;
         end
       end
       default: begin
         state <= ST_IDLE;
       end
     endcase
   end
 end

 assign s_arready = (state == ST_IDLE) && !(aw_have && w_have) && !(aw_have && s_wvalid) && !(w_have && s_awvalid);

endmodule
