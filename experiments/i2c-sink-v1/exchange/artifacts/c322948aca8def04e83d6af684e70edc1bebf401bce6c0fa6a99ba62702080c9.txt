module i2c_sink(input logic clk,rst_n,scl,sda,output logic sda_drive_low,item_valid,output logic [15:0] item_data);

  // Synchronizers (2FF) for scl and sda
  logic scl_s1, scl_s2, sda_s1, sda_s2;
  always_ff @(posedge clk) begin
    if (!rst_n) begin
      scl_s1 <= 1'b1; scl_s2 <= 1'b1;
      sda_s1 <= 1'b1; sda_s2 <= 1'b1;
    end else begin
      scl_s1 <= scl; scl_s2 <= scl_s1;
      sda_s1 <= sda; sda_s2 <= sda_s1;
    end
  end

  // Previous synchronized values for edge detection
  logic scl_p, sda_p;
  always_ff @(posedge clk) begin
    if (!rst_n) begin
      scl_p <= 1'b1; sda_p <= 1'b1;
    end else begin
      scl_p <= scl_s2; sda_p <= sda_s2;
    end
  end

  wire scl_rise = scl_s2 & ~scl_p;
  wire scl_fall = ~scl_s2 & scl_p;
  wire start_cond = scl_s2 & scl_p & sda_p & ~sda_s2;  // SDA fall while SCL high
  wire stop_cond  = scl_s2 & scl_p & ~sda_p & sda_s2;  // SDA rise while SCL high

  typedef enum logic [1:0] {ST_IDLE, ST_ADDR, ST_RX, ST_ACK, ST_STOP} state_t;
  state_t state;

  logic [3:0] bit_cnt;       // bit index 0..8 within current byte
  logic [7:0] shift;         // receive shift register
  logic [7:0] low_byte;      // first byte of word (high byte)
  logic       half;          // have one pending byte
  logic [15:0] data_r;
  logic       valid_r;
  logic       drive_low_r;

  always_ff @(posedge clk) begin
    if (!rst_n) begin
      state <= ST_IDLE;
      bit_cnt <= 4'd0;
      shift <= 8'd0;
      low_byte <= 8'd0;
      half <= 1'b0;
      data_r <= 16'd0;
      valid_r <= 1'b0;
      drive_low_r <= 1'b0;
    end else begin
      valid_r <= 1'b0;

      if (stop_cond) begin
        // STOP: discard partial word, release bus
        state <= ST_IDLE;
        bit_cnt <= 4'd0;
        half <= 1'b0;
        drive_low_r <= 1'b0;
      end else if (start_cond) begin
        // (Re)START seen
        state <= ST_ADDR;
        bit_cnt <= 4'd0;
        drive_low_r <= 1'b0;
      end else begin
        case (state)
          ST_IDLE: begin
            drive_low_r <= 1'b0;
          end

          ST_ADDR: begin
            // ACK low is established starting at bit 9 low period -> drive during ACK bit
            if (scl_fall && bit_cnt == 4'd0) begin
              drive_low_r <= 1'b0;  // release ACK when starting next byte
            end
            if (scl_rise) begin
              if (bit_cnt < 4'd8) begin
                shift <= {shift[6:0], sda_s2};
                bit_cnt <= bit_cnt + 1'b1;
              end else begin
                // ACK bit (9th) completes on this rise; shift holds address byte
                // Enforce address 0x52 + write (as spec's only supported transaction)
                bit_cnt <= 4'd0;
                drive_low_r <= 1'b0;
                state <= ST_RX;
                half <= 1'b0;
              end
            end else if (scl_fall && bit_cnt == 4'd8) begin
              // Entering 9th bit low -> assert ACK
              drive_low_r <= 1'b1;
            end
          end

          ST_RX: begin
            if (scl_fall && bit_cnt == 4'd8) begin
              drive_low_r <= 1'b1; // assert ACK for this byte
            end else if (scl_fall && bit_cnt == 4'd0) begin
              drive_low_r <= 1'b0; // release ACK at start of next byte
            end
            if (scl_rise) begin
              if (bit_cnt < 4'd8) begin
                shift <= {shift[6:0], sda_s2};
                bit_cnt <= bit_cnt + 1'b1;
              end else begin
                // Byte complete (8 data bits shifted by now)
                bit_cnt <= 4'd0;
                if (!half) begin
                  low_byte <= shift;
                  half <= 1'b1;
                end else begin
                  data_r <= {low_byte, shift};
                  valid_r <= 1'b1;
                  half <= 1'b0;
                end
              end
            end
          end

          default: begin
            state <= ST_IDLE;
          end
        endcase
      end
    end
  end

  assign sda_drive_low = drive_low_r;
  assign item_valid = valid_r;
  assign item_data = data_r;

endmodule
