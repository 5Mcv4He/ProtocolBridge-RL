module swd_read(input wire clk,rst_n,swclk,swdio,input wire [31:0] read_value,output wire dout,oe);

  reg swing;
  reg [3:0] bitcnt;
  reg req_active;
  reg in_txn;
  reg [5:0] phase;       // 1..38
  reg [31:0] data_sh;
  reg dout_r;
  reg oe_r;

  assign dout = dout_r;
  assign oe = oe_r;

  wire rise  = swclk & ~swing;
  wire fall  = ~swclk & swing;

  always @(posedge clk) begin
    if (!rst_n) begin
      swing <= 1'b0;
      bitcnt <= 4'd0;
      req_active <= 1'b0;
      in_txn <= 1'b0;
      phase <= 6'd0;
      data_sh <= 32'd0;
      dout_r <= 1'b0;
      oe_r <= 1'b0;
    end else begin
      swing <= swclk;

      if (fall) begin
        // Release the line before the master samples / turns around.
        oe_r <= 1'b0;
        dout_r <= 1'b0;
      end

      if (rise) begin
        if (!in_txn) begin
          if (!req_active) begin
            if (swdio) begin
              bitcnt <= 4'd1;
              req_active <= 1'b1;
            end
          end else begin
            if (bitcnt == 4'd7) begin
              req_active <= 1'b0;
              bitcnt <= 4'd0;
              in_txn <= 1'b1;
              phase <= 6'd1;
              data_sh <= read_value;
            end else begin
              bitcnt <= bitcnt + 4'd1;
            end
          end
        end else begin
          if (phase == 6'd38) begin
            oe_r <= 1'b0;
            dout_r <= 1'b0;
            in_txn <= 1'b0;
            phase <= 6'd0;
          end else begin
            // drive bit for the interval starting at this rising edge
            oe_r <= 1'b1;
            if (phase == 6'd2 || phase == 6'd3)
              dout_r <= 1'b0;
            else if (phase >= 6'd1)
              dout_r <= (phase == 6'd4) ? data_sh[0] : data_sh[0];
            if (phase == 6'd2)
              dout_r <= 1'b1;
            if (phase >= 6'd4)
              data_sh <= {1'b0, data_sh[31:1]};
            if (phase == 6'd37)
              dout_r <= ^data_sh;
            // ACK bits: 2=1, 3=0, 4=0
            if (phase == 6'd1) dout_r <= 1'b0;
            phase <= phase + 6'd1;
          end
        end
      end
    end
  end
endmodule
