module fifo_checker(input logic clk,rst_n,push,pop,
 input logic [7:0] push_id,pop_id,output logic error);
  logic [7:0] q [0:4];
  logic [2:0] count;
  integer i;

  always_ff @(posedge clk) begin
    if (!rst_n) begin
      error <= 1'b0;
      count <= 3'd0;
      for (i = 0; i < 5; i = i + 1) q[i] <= 8'd0;
    end else if (!error) begin
      if (pop && (count == 3'd0 || pop_id != q[0])) begin
        error <= 1'b1;
      end else if (push && !pop && count == 3'd5) begin
        error <= 1'b1;
      end else begin
        if (push && pop) begin
          for (i = 0; i < 4; i = i + 1) q[i] <= q[i+1];
          q[count - 3'd1] <= push_id;
          count <= count;
        end else if (push) begin
          q[count] <= push_id;
          count <= count + 3'd1;
        end else if (pop) begin
          for (i = 0; i < 4; i = i + 1) q[i] <= q[i+1];
          count <= count - 3'd1;
        end
      end
    end
  end
endmodule
