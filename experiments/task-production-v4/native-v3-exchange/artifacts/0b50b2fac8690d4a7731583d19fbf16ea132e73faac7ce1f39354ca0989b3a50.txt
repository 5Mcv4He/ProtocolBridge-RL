module rffe_register(
  input  logic clk,
  input  logic rst_n,
  input  logic sclk,
  input  logic sdata,
  output logic sdata_out,
  output logic sdata_oe
);

  // RFFE register slave with W1C semantics.
  // Registers: 16 x 8-bit, reset to 255.
  logic [7:0] regs [0:15];

  // Synchronizers for async pins (2-stage each)
  logic sclk_meta, sclk_sync;
  logic sdat_meta, sdat_sync;
  logic sclk_d, sdat_d;

  // Command decoder state
  typedef enum logic [2:0] {
    ST_IDLE,      // waiting for SSC
    ST_USID,      // collecting 4-bit USID
    ST_CMD,       // collecting 8-bit command
    ST_CPAR,      // command parity
    ST_WDATA,     // collecting 8-bit write data
    ST_WPAR,      // write data parity
    ST_WPARK,     // master bus park after write
    ST_RPARK,     // master bus park before read data
    ST_RDATA,     // shifting out read data
    ST_RPAR,      // shifting out parity bit
    ST_REL        // release after read bus park
  } state_t;

  state_t state;
  logic [3:0] bitcnt;
  logic [3:0] usid;
  logic [7:0] cmd;
  logic [3:0] addr;
  logic [7:0] wdata;
  logic [7:0] rdata;
  logic par_acc;
  logic is_write;
  logic [3:0] tail_cnt;

  wire sclk_rise = (sclk_sync == 1'b1) && (sclk_d == 1'b0);
  wire sclk_fall = (sclk_sync == 1'b0) && (sclk_d == 1'b1);

  integer i;

  always_ff @(posedge clk) begin
    if (!rst_n) begin
      sclk_meta <= 1'b0;
      sclk_sync <= 1'b0;
      sdat_meta <= 1'b0;
      sdat_sync <= 1'b0;
      sclk_d    <= 1'b0;
      sdat_d    <= 1'b0;
      state     <= ST_IDLE;
      bitcnt    <= 4'd0;
      usid      <= 4'd0;
      cmd       <= 8'd0;
      addr      <= 4'd0;
      wdata     <= 8'd0;
      rdata     <= 8'd0;
      par_acc   <= 1'b0;
      is_write  <= 1'b0;
      tail_cnt  <= 4'd0;
      sdata_out <= 1'b0;
      sdata_oe  <= 1'b0;
      for (i = 0; i < 16; i = i + 1)
        regs[i] <= 8'hFF;
    end else begin
      // Synchronizers
      sclk_meta <= sclk;
      sclk_sync <= sclk_meta;
      sdat_meta <= sdata;
      sdat_sync <= sdat_meta;
      sclk_d    <= sclk_sync;
      sdat_d    <= sdat_sync;

      // Detect SSC: SDATA high pulse while SCLK stays low.
      // We register a flag when a high level observed on sdata while sclk low
      // then on next rising edge body enters USID. Simpler: at sclk rising edge,
      // if currently IDLE and sdata sampled high, it's the SSC.
      if (sclk_rise) begin
        case (state)
          ST_IDLE: begin
            if (sdat_sync) begin
              // SSC detected
              state   <= ST_USID;
              bitcnt  <= 4'd0;
              par_acc <= 1'b0;
              usid    <= 4'd0;
              cmd     <= 8'd0;
            end
          end
          ST_USID: begin
            usid    <= {usid[2:0], sdat_sync};
            bitcnt  <= bitcnt + 4'd1;
            if (bitcnt == 4'd3) begin
              state  <= ST_CMD;
              bitcnt <= 4'd0;
            end
          end
          ST_CMD: begin
            cmd     <= {cmd[6:0], sdat_sync};
            par_acc <= par_acc ^ sdat_sync;
            bitcnt  <= bitcnt + 4'd1;
            if (bitcnt == 4'd7) begin
              state  <= ST_CPAR;
              bitcnt <= 4'd0;
            end
          end
          ST_CPAR: begin
            if (is_write) begin
              state  <= ST_WDATA;
              bitcnt <= 4'd0;
              wdata  <= 8'd0;
              par_acc <= 1'b0;
            end else begin
              state  <= ST_RPARK;
              bitcnt <= 4'd0;
            end
          end
          ST_WDATA: begin
            wdata   <= {wdata[6:0], sdat_sync};
            par_acc <= par_acc ^ sdat_sync;
            bitcnt  <= bitcnt + 4'd1;
            if (bitcnt == 4'd7) begin
              state  <= ST_WPAR;
              bitcnt <= 4'd0;
            end
          end
          ST_WPAR: begin
            // apply write
            for (i = 0; i < 16; i = i + 1) begin
              if (i == addr) begin
                regs[i] <= regs[i] & ~wdata;
              end
            end
            state  <= ST_WPARK;
          end
          ST_RPARK: begin
            // master bus park bit completed at this rising edge;
            // start driving read data from next falling edge onward.
            rdata <= regs[addr];
            state <= ST_RDATA;
            bitcnt <= 4'd0;
          end
          default: begin
            state <= ST_IDLE;
          end
        endcase
      end

      // Determine is_write/addr from command once complete (on CPAR rising edge
      // cmd has its last bit updated; compute combinational below instead).
      // Handle outputs on falling edges.
      if (sclk_fall) begin
        case (state)
          ST_RDATA: begin
            // Output next bit; set at falling edge so valid before next rising.
            sdata_out <= rdata[7];
            sdata_oe  <= 1'b1;
          end
          ST_RPAR: begin
            sdata_out <= par_acc;
            sdata_oe  <= 1'b1;
          end
          default: begin
            sdata_oe  <= 1'b0;
            sdata_out <= 1'b0;
          end
        endcase
      end
    end
  end

  // is_write / addr logic based on command value (upper bits):
  // write: cmd[7:6] == 01, read: cmd[7:6] == 10, addr = cmd[3:0].
  always_comb begin
    is_write = (cmd[7:6] == 2'b01);
    addr     = cmd[3:0];
  end

endmodule
