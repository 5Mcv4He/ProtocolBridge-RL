module rffe_register(
  input  logic clk,
  input  logic rst_n,
  input  logic sclk,
  input  logic sdata,
  output logic sdata_out,
  output logic sdata_oe
);

  // RFFE register slave with W1C semantics.
  // 16 x 8-bit registers, reset to 255.
  logic [7:0] regs [0:15];

  // Pin synchronizers (2-stage)
  logic sclk_meta, sclk_sync;
  logic sdat_meta, sdat_sync;
  logic sclk_d;

  typedef enum logic [3:0] {
    ST_IDLE,
    ST_USID,
    ST_CMD,
    ST_CPAR,
    ST_WDATA,
    ST_WPAR,
    ST_WPARK,
    ST_RPARK,
    ST_RBIT,
    ST_RPAR,
    ST_RPARK2,
    ST_REL
  } state_t;

  state_t state;
  logic [3:0] bitcnt;
  logic [7:0] cmd;
  logic [7:0] wdata;
  logic [7:0] shift_r;
  logic par_acc;
  logic is_write;
  logic [3:0] addr;

  wire sclk_rise = (sclk_sync == 1'b1) && (sclk_d == 1'b0);
  wire sclk_fall = (sclk_sync == 1'b0) && (sclk_d == 1'b1);

  integer i;

  always_ff @(posedge clk) begin
    if (!rst_n) begin
      sclk_meta <= 1'b0;
      sclk_sync <= 1'b0;
      sdat_meta <= 1'b0;
      sdat_sync <= 1'b0;
      sclk_d    <= 1'b0;
      state     <= ST_IDLE;
      bitcnt    <= 4'd0;
      cmd       <= 8'd0;
      wdata     <= 8'd0;
      shift_r   <= 8'd0;
      par_acc   <= 1'b0;
      is_write  <= 1'b0;
      addr      <= 4'd0;
      sdata_out <= 1'b0;
      sdata_oe  <= 1'b0;
      for (i = 0; i < 16; i = i + 1)
        regs[i] <= 8'hFF;
    end else begin
      sclk_meta <= sclk;
      sclk_sync <= sclk_meta;
      sdat_meta <= sdata;
      sdat_sync <= sdat_meta;
      sclk_d    <= sclk_sync;

      if (sclk_rise) begin
        case (state)
          ST_IDLE: begin
            if (sdat_sync) begin
              // SSC detected
              state   <= ST_USID;
              bitcnt  <= 4'd0;
              par_acc <= 1'b0;
              cmd     <= 8'd0;
            end
          end
          ST_USID: begin
            bitcnt <= bitcnt + 4'd1;
            if (bitcnt == 4'd3) begin
              state   <= ST_CMD;
              bitcnt  <= 4'd0;
              par_acc <= 1'b0;
            end
          end
          ST_CMD: begin
            cmd     <= {cmd[6:0], sdat_sync};
            par_acc <= par_acc ^ sdat_sync;
            bitcnt  <= bitcnt + 4'd1;
            if (bitcnt == 4'd7) begin
              state  <= ST_CPAR;
              bitcnt <= 4'd0;
            end
          end
          ST_CPAR: begin
            is_write <= (cmd[7:6] == 2'b01);
            addr     <= cmd[3:0];
            if (cmd[7:6] == 2'b01) begin
              state   <= ST_WDATA;
              bitcnt  <= 4'd0;
              wdata   <= 8'd0;
              par_acc <= 1'b0;
            end else begin
              state   <= ST_RPARK;
              bitcnt  <= 4'd0;
            end
          end
          ST_WDATA: begin
            wdata   <= {wdata[6:0], sdat_sync};
            par_acc <= par_acc ^ sdat_sync;
            bitcnt  <= bitcnt + 4'd1;
            if (bitcnt == 4'd7) begin
              state  <= ST_WPAR;
              bitcnt <= 4'd0;
            end
          end
          ST_WPAR: begin
            if (addr <= 4'd15)
              regs[addr] <= regs[addr] & ~wdata;
            state <= ST_WPARK;
          end
          ST_WPARK: begin
            state <= ST_IDLE;
          end
          ST_RPARK: begin
            // master bus park bit sampled; load read data
            shift_r <= regs[addr];
            state   <= ST_RBIT;
            bitcnt  <= 4'd1;
            par_acc <= regs[addr][7];
          end
          ST_RBIT: begin
            par_acc <= par_acc ^ sdata_out;
            shift_r <= {shift_r[6:0], 1'b0};
            bitcnt  <= bitcnt + 4'd1;
            if (bitcnt == 4'd8) begin
              state  <= ST_RPAR;
              bitcnt <= 4'd0;
            end
          end
          ST_RPAR: begin
            state <= ST_RPARK2;
          end
          ST_RPARK2: begin
            state <= ST_REL;
          end
          ST_REL: begin
            sdata_oe  <= 1'b0;
            sdata_out <= 1'b0;
            state     <= ST_IDLE;
          end
          default: state <= ST_IDLE;
        endcase
      end

      if (sclk_fall) begin
        case (state)
          ST_RPARK: begin
            // data bit 7 must be set up before this falling edge
            sdata_out <= regs[addr][7];
            sdata_oe  <= 1'b1;
          end
          ST_RBIT: begin
            sdata_out <= shift_r[6];
            sdata_oe  <= 1'b1;
          end
          ST_RPAR: begin
            sdata_out <= ~par_acc;
            sdata_oe  <= 1'b1;
          end
          ST_RPARK2: begin
            // candidate bus park; release after falling edge
            sdata_out <= 1'b0;
            sdata_oe  <= 1'b0;
          end
          default: begin
            sdata_oe  <= 1'b0;
            sdata_out <= 1'b0;
          end
        endcase
      end

      // Safety release in IDLE (idle must not drive)
      if (state == ST_IDLE) begin
        sdata_oe  <= 1'b0;
        sdata_out <= 1'b0;
      end
    end
  end

endmodule
