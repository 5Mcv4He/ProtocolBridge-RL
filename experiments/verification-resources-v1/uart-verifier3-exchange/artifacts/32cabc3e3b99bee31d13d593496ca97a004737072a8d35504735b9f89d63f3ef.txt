module uart_tx(input logic clk,rst_n,valid,input logic [7:0] data,
               output logic ready,tx);
  logic [9:0] frame;
  logic [3:0] bit_cnt, tick;

  assign ready = (bit_cnt == 4'd0);
  assign tx = ready ? 1'b1 : frame[9];

  always_ff @(posedge clk) begin
    if (!rst_n) begin
      frame   <= 10'h3FF;
      bit_cnt <= 4'd0;
      tick    <= 4'd0;
    end else if (ready) begin
      if (valid) begin
        frame   <= {1'b0, data[7:0], 1'b1};
        bit_cnt <= 4'd10;
        tick    <= 4'd0;
      end
    end else if (tick == 4'd15) begin
      tick    <= 4'd0;
      frame   <= {frame[8:0], 1'b1};
      bit_cnt <= bit_cnt - 4'd1;
    end else begin
      tick <= tick + 4'd1;
    end
  end
endmodule
