module uart_tx(input logic clk,rst_n,valid,input logic [7:0] data,
               output logic ready,tx);
  logic [9:0] frame;
  logic [9:0] frame_q;
  logic [3:0] remaining,tick;
  logic busy;

  assign busy = (remaining != 4'd0);
  assign ready = !busy;
  assign tx = busy ? frame_q[9] : 1'b1;

  always_ff @(posedge clk) begin
    if (!rst_n) begin
      frame_q <= 10'h3FF;
      remaining <= 4'd0;
      tick <= 4'd0;
    end else if (!busy) begin
      tick <= 4'd0;
      if (valid) begin
        frame_q <= {1'b1, data, 1'b0};
        remaining <= 4'd10;
      end
    end else if (tick == 4'd15) begin
      tick <= 4'd0;
      if (remaining == 4'd1) begin
        remaining <= 4'd0;
        frame_q <= 10'h3FF;
      end else begin
        remaining <= remaining - 4'd1;
        frame_q <= {frame_q[8:0], 1'b1};
      end
    end else begin
      tick <= tick + 4'd1;
    end
  end
endmodule
