module uart_tx(input logic clk,rst_n,valid,input logic [7:0] data, output logic ready,tx);
  logic [9:0] sh;
  logic [3:0] bitcnt;
  logic [3:0] tick;
  logic busy;

  assign busy = (bitcnt != 4'd0);
  assign ready = !busy;
  assign tx = busy ? sh[0] : 1'b1;

  always_ff @(posedge clk) begin
    if (!rst_n) begin
      sh <= 10'h3FF;
      bitcnt <= 4'd0;
      tick <= 4'd0;
    end else if (!busy) begin
      tick <= 4'd0;
      if (valid) begin
        sh <= {1'b1, data, 1'b0};
        bitcnt <= 4'd10;
      end
    end else if (tick == 4'd15) begin
      tick <= 4'd0;
      if (bitcnt == 4'd1) begin
        bitcnt <= 4'd0;
        sh <= 10'h3FF;
      end else begin
        bitcnt <= bitcnt - 4'd1;
        sh <= {1'b1, sh[9:1]};
      end
    end else begin
      tick <= tick + 4'd1;
    end
  end
endmodule
