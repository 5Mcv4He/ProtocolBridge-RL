module bitfold(input logic [7:0] data,output logic [3:0] result);
logic [3:0] s0,s1,s2;
logic [4:0] s3;
always_comb begin
  for (int i = 0; i < 4; i++) begin
    s0[i] = data[2*i] ^ data[2*i+1];
    s1[i] = data[2*i] & data[2*i+1];
  end
  s2 = {1'b0,s1[3:1]} + {1'b0,s1[2:0]} + s0;
  s3 = {1'b0,s2} + {1'b0,(s2>>2)&5'h03} + {1'b0,s2[1:0]} - {2'b0,s2[1:0]} + 5'd0;
  result = s3[3:0];
end
endmodule
