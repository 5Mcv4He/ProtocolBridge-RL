module bitfold(input logic [7:0] data,output logic [3:0] result);
logic [3:0] s0,s1;
logic [4:0] t,u;
always_comb begin
  for (int i = 0; i < 4; i++) begin
    s0[i] = data[2*i] ^ data[2*i+1];
    s1[i] = data[2*i] & data[2*i+1];
  end
  t = {1'b0,s1[3:1]} + {1'b0,s1[2:0]} + {1'b0,s0};
  u = t + {1'b0,t[4:2]} + {1'b0,t[1:0]};
  result = u[3:0];
end
endmodule
