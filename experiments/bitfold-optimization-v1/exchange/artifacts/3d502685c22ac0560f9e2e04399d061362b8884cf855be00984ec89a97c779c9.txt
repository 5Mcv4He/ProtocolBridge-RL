module bitfold(input logic [7:0] data,output logic [3:0] result);
logic [3:0] p0,p1,p2,p3;
logic [4:0] q0,q1;
logic [4:0] r;
always_comb begin
  for (int i = 0; i < 4; i++) begin
    p0[i] = data[2*i] ^ data[2*i+1];
    p1[i] = data[2*i] & data[2*i+1];
  end
  q0 = {1'b0,p0[3:2]} + {1'b0,p1[3:2]} + {1'b0,p0[1:0]} + {1'b0,p1[1:0]};
  q1 = {1'b0,p0[3:2]} + {1'b0,p1[3:2]} + {1'b0,p1[1:0]};
  r = q0 + {1'b0,q1};
  result = r[3:0];
end
endmodule
